        function createStars() {

            const container = document.getElementById('stars');

            const count = 120;

            const w = window.innerWidth;

            const h = window.innerHeight;

            const maxRadius = Math.sqrt(Math.pow(w/2, 2) + Math.pow(h/2, 2));


            for (let i = 0; i < count; i++) {

                const star = document.createElement('div');

                star.className = 'star';

                const angle = Math.random() * Math.PI * 2;

                const destX = Math.cos(angle) * maxRadius;

                const destY = Math.sin(angle) * maxRadius;

                const isHero = Math.random() < 0.05;

                const size = isHero ? (1 + Math.random() * 0.75) : (0.4 + Math.random() * 0.6);

                const duration = isHero ? (1.5 + Math.random() * 1.5) : (5 + Math.random() * 5); 

                const delay = Math.random() * -10; 

                const endScale = isHero ? 3.5 : 1.5;

                star.style.width = `${size}px`;

                star.style.height = `${size}px`;

                star.style.setProperty('--destX', `${destX}px`);

                star.style.setProperty('--destY', `${destY}px`);

                star.style.setProperty('--duration', `${duration}s`);

                star.style.setProperty('--delay', `${delay}s`);

                star.style.setProperty('--endScale', endScale);

                container.appendChild(star);

            }

        }


        const paths = DOJO_DATA.unit1.path7;

        let activePath = null;

        let index = 0;

        let streak = 0;


        function renderGrid() {

            document.getElementById('pathGrid').innerHTML = paths.map((p, i) => `

                <div class="path-card p-8 rounded-[2.5rem] group" onclick="startPath(${i})">

                    <div class="w-16 h-16 bg-purple-500/20 text-purple-400 rounded-2xl flex items-center justify-center mb-6 group-hover:bg-purple-500 group-hover:text-white transition-all shadow-lg">

                        <i class="fa-solid fa-route text-3xl"></i>

                    </div>

                    <h3 class="text-2xl font-black text-white mb-2 uppercase tracking-tight">${p.title}</h3>

                    <p class="text-slate-400 text-xs font-medium leading-relaxed">${p.description}</p>

                </div>

            `).join('');

        }


        function startPath(i) {

            activePath = paths[i];

            index = 0;

            document.getElementById('selectionStage').classList.add('hidden');

            document.getElementById('practiceStage').classList.remove('hidden');

            document.getElementById('practiceTitle').innerText = activePath.title;

            document.getElementById('practiceFormula').innerText = `FORMULA: ${activePath.formula}`;

            loadDrill();

        }


        function loadDrill() {

            document.getElementById('practiceWeak').innerText = `"${activePath.drills[index]}"`;

            document.getElementById('userInput').value = "";

            document.getElementById('inputArea').classList.remove('hidden');

            document.getElementById('successArea').classList.add('hidden');

            setTimeout(() => document.getElementById('userInput').focus(), 100);

        }


        function forge() {

            const val = document.getElementById('userInput').value.trim();

            if(!val) return;

            document.getElementById('practiceWeak').innerHTML = `<span class="forged-text">"${val}"</span>`;

            document.getElementById('inputArea').classList.add('hidden');

            document.getElementById('successArea').classList.remove('hidden');

            streak++;

            document.getElementById('streakCounter').innerText = streak;

        }


        function nextDrill() {

            index++;

            if (index < activePath.drills.length) loadDrill();

            else showMaster();

        }


        function showMaster() {

            document.getElementById('practiceStage').classList.add('hidden');

            document.getElementById('masterStage').classList.remove('hidden');

            document.getElementById('paragraphContent').innerHTML = activePath.paragraph;

            const cover = document.getElementById('ninjaCover');

            cover.classList.remove('hidden');

            setTimeout(() => {

                cover.classList.add('animate-slice-open');

                setTimeout(() => {

                    cover.classList.add('hidden');

                    document.getElementById('masterInputArea').classList.remove('hidden');

                    document.getElementById('masterInput').focus();

                }, 2500);

            }, 500);

        }


        function forgeMaster() {

            const val = document.getElementById('masterInput').value.trim();

            if (!val) return;

            const target = document.getElementById('p-target');

            target.innerHTML = val;

            target.className = "text-amber-400 font-black italic drop-shadow-md underline decoration-amber-500/30 underline-offset-8";

            document.getElementById('masterInputArea').classList.add('hidden');

            document.getElementById('completionArea').classList.remove('hidden');

        }


        window.onload = function() { createStars(); renderGrid(); };

        window.onresize = () => { document.getElementById('stars').innerHTML = ''; createStars(); };
