        function createStars() {

            const container = document.getElementById('stars');

            const count = 120;

            const w = window.innerWidth;

            const h = window.innerHeight;

            const maxRadius = Math.sqrt(Math.pow(w/2, 2) + Math.pow(h/2, 2));


            for (let i = 0; i < count; i++) {

                const star = document.createElement('div');

                star.className = 'star';

                const angle = Math.random() * Math.PI * 2;

                const destX = Math.cos(angle) * maxRadius;

                const destY = Math.sin(angle) * maxRadius;

                const isHero = Math.random() < 0.05;

                const size = isHero ? (1 + Math.random() * 0.75) : (0.4 + Math.random() * 0.6);

                const duration = isHero ? (1.5 + Math.random() * 1.5) : (5 + Math.random() * 5); 

                const delay = Math.random() * -10; 

                const endScale = isHero ? 3.5 : 1.5;

                star.style.width = `${size}px`;

                star.style.height = `${size}px`;

                star.style.setProperty('--destX', `${destX}px`);

                star.style.setProperty('--destY', `${destY}px`);

                star.style.setProperty('--duration', `${duration}s`);

                star.style.setProperty('--delay', `${delay}s`);

                star.style.setProperty('--endScale', endScale);

                container.appendChild(star);

            }

        }


        // Pull from Master Brain (Defaulting to Unit 1 for prototype)

        const unitData = typeof DOJO_DATA !== 'undefined' ? DOJO_DATA.unit1 : { assassin: { targets: [], paragraph: "" } };

        const assassinData = unitData.assassin;


        let streak = 0;

        let currentIndex = -1;

        let isParagraphMode = false;

        let activeTargetSpan = null;


        const sentenceContainer = document.getElementById('sentenceContainer');

        const paragraphContainer = document.getElementById('paragraphContainer');

        const interactionArea = document.getElementById('interactionArea');

        const nextArea = document.getElementById('nextArea');

        const userInput = document.getElementById('userInput');

        const hintArea = document.getElementById('hintArea');


        function init() {

            createStars();

            loadNext();

            userInput.addEventListener("keypress", (e) => e.key === "Enter" && strike());

        }


        function loadNext() {

            currentIndex++;

            if (currentIndex >= assassinData.targets.length) currentIndex = 0;

            const item = assassinData.targets[currentIndex];

            sentenceContainer.innerHTML = `<span id="target" class="weak-word px-2">${item.prefix}${item.weak}</span><span>${item.suffix}</span>`;

            activeTargetSpan = document.getElementById('target');

            document.getElementById('targetWordLabel').innerText = item.target;

            userInput.value = '';

            interactionArea.classList.remove('hidden');

            nextArea.classList.add('hidden');

            hintArea.classList.add('hidden');

            document.getElementById('cheatSheetBtnText').innerText = 'Show Cheat Sheet';

            setTimeout(() => userInput.focus(), 100);

        }


        function strike() {

            const val = userInput.value.trim();

            if (!val || !activeTargetSpan) return;

            const rect = activeTargetSpan.getBoundingClientRect();

            const arrow = document.getElementById('flyingArrow');

            arrow.style.left = `${rect.left + (rect.width/2) - 25}px`;

            arrow.style.top = `${rect.top - 20}px`;

            arrow.classList.remove('hidden', 'arrow-animating');

            void arrow.offsetWidth;

            arrow.classList.add('arrow-animating');


            setTimeout(() => {

                arrow.classList.add('hidden');

                activeTargetSpan.innerText = val;

                activeTargetSpan.className = 'word-assassinated px-2';

                streak++;

                document.getElementById('streakCounter').innerText = streak;

                interactionArea.classList.add('hidden');

                if (isParagraphMode) handleParagraphProgress();

                else if (streak === 5) showScrollUnlock();

                else showSuccess();

            }, 400);

        }


        function showSuccess() {

            nextArea.innerHTML = `

                <div class="text-3xl font-black text-rose-400 italic uppercase">Target Eliminated!</div>

                <button onclick="loadNext()" class="bg-slate-800 hover:bg-slate-700 text-white border border-white/10 rounded-2xl px-12 py-5 text-2xl font-bold uppercase flex items-center gap-3 transition">

                    Next Target <i class="fa-solid fa-arrow-right"></i>

                </button>

            `;

            nextArea.classList.remove('hidden');

            nextArea.classList.add('flex');

        }


        function showScrollUnlock() {

            const celeb = document.getElementById('celebrationOverlay');

            document.getElementById('celebrationText').innerHTML = `Scroll Unlocked!<br><span class="text-4xl italic">Master Assassin Mode</span>`;

            celeb.classList.remove('hidden');

            celeb.firstElementChild.classList.add('animate-epic-pop');

            setTimeout(() => celeb.classList.add('hidden'), 3000);

            document.getElementById('textDisplayArea').classList.add('hidden');

            nextArea.innerHTML = `

                <div class="text-3xl font-black text-amber-400 mb-2 italic">You are ready.</div>

                <button onclick="startMasterScroll()" class="bg-amber-600 hover:bg-amber-500 text-white border-b-8 border-amber-800 rounded-[2rem] px-12 py-5 text-2xl font-bold uppercase flex items-center gap-3 shadow-2xl">

                    <i class="fa-solid fa-scroll"></i> Open Master Scroll

                </button>

            `;

            nextArea.classList.remove('hidden');

            nextArea.classList.add('flex');

        }


        function startMasterScroll() {

            isParagraphMode = true;

            document.getElementById('textDisplayArea').classList.remove('hidden');

            sentenceContainer.classList.add('hidden');

            document.getElementById('paragraphWrapper').classList.remove('hidden');

            document.getElementById('modeTitle').innerText = "Assassin Scroll: Direct Action";

            document.getElementById('modeTitle').classList.replace('text-rose-500', 'text-amber-400');

            

            let idCount = 0;

            const html = assassinData.paragraph.replace(/\{([^}]+)\}/g, (match, word) => {

                const id = `p_word_${idCount++}`;

                return `<span id="${id}" class="weak-word px-2" onclick="selectParagraphWord('${id}', '${word}')">${word}</span>`;

            });

            paragraphContainer.innerHTML = html;

            

            const cover = document.getElementById('ninjaCover');

            cover.classList.remove('hidden', 'animate-slice-open');

            void cover.offsetWidth;

            cover.classList.add('animate-slice-open');

            setTimeout(() => {

                cover.classList.add('hidden');

                selectParagraphWord('p_word_0', paragraphContainer.querySelector('.weak-word').innerText);

            }, 2600);

        }


        function selectParagraphWord(id, original) {

            if (activeTargetSpan) activeTargetSpan.classList.remove('weak-word-active');

            activeTargetSpan = document.getElementById(id);

            activeTargetSpan.classList.add('weak-word-active');

            document.getElementById('targetWordLabel').innerText = original;

            userInput.value = '';

            interactionArea.classList.remove('hidden');

            nextArea.classList.add('hidden');

            setTimeout(() => userInput.focus(), 100);

        }


        function handleParagraphProgress() {

            const nextWeak = paragraphContainer.querySelector('.weak-word');

            if (nextWeak) setTimeout(() => selectParagraphWord(nextWeak.id, nextWeak.innerText), 600);

            else {

                nextArea.innerHTML = `

                    <div class="text-4xl font-black text-amber-400 italic">Mission Accomplished!</div>

                    <button onclick="window.location.href='dashboard.html'" class="bg-slate-800 hover:bg-slate-700 text-white border border-white/10 rounded-2xl px-12 py-5 text-2xl font-bold uppercase mt-4 transition">Return to Dojo</button>

                `;

                nextArea.classList.remove('hidden');

                nextArea.classList.add('flex');

            }

        }


        function toggleCheatSheet() {

            const btnText = document.getElementById('cheatSheetBtnText');

            if (hintArea.classList.contains('hidden')) {

                hintArea.classList.remove('hidden');

                btnText.innerText = 'Hide Cheat Sheet';

            } else {

                hintArea.classList.add('hidden');

                btnText.innerText = 'Show Cheat Sheet';

            }

        }


        window.onload = init;

        window.onresize = () => { document.getElementById('stars').innerHTML = ''; createStars(); };
