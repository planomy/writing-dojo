    <script src="content.js"></script>

    <script>

        // DATA FALLBACK if content.js isn't loaded correctly

        const KATA_DATA = typeof DOJO_DATA !== 'undefined' ? DOJO_DATA.unit1.katana : [

            { title: "Triple Punchers", formula: "Action. Action. Impact.", description: "Banish rambling. Strike with rhythmic speed.", drills: ["He ran and jumped and fell.", "The guard saw him and shouted and ran."], paragraph: "The night was quiet. <span id='p-target' class='bg-rose-100/20 text-rose-400 px-2'>The ninja moved and then he stopped and then he hid.</span>" }

        ];


        let activeKata = null;

        let drillIndex = 0;

        let streak = 0;


        function renderGrid() {

            const grid = document.getElementById('kataGrid');

            grid.innerHTML = KATA_DATA.map((k, i) => `

                <div class="kata-card p-10 rounded-[2.5rem] group" onclick="startPractice(${i})">

                    <div class="flex justify-between mb-8">

                        <div class="w-16 h-16 bg-blue-500/20 text-blue-400 rounded-2xl flex items-center justify-center group-hover:bg-blue-500 group-hover:text-white transition-all shadow-lg group-hover:rotate-12">

                            <i class="fa-solid fa-pen-ruler text-3xl"></i>

                        </div>

                        <div class="text-[10px] font-black uppercase tracking-widest text-slate-500 group-hover:text-blue-400 transition-colors">Blueprint</div>

                    </div>

                    <h3 class="text-2xl font-black text-white mb-3 uppercase tracking-tight">${k.title}</h3>

                    <p class="text-slate-400 leading-relaxed text-sm font-medium">${k.description}</p>

                </div>

            `).join('');

        }


        function startPractice(i) {

            activeKata = KATA_DATA[i];

            drillIndex = 0;

            document.getElementById('selectionStage').classList.add('hidden');

            document.getElementById('practiceStage').classList.remove('hidden');

            document.getElementById('practiceKataTitle').innerText = activeKata.title;

            document.getElementById('practiceFormula').innerText = `FORMULA: ${activeKata.formula}`;

            loadDrill();

        }


        function loadDrill() {

            document.getElementById('practiceProgress').innerText = `Target ${drillIndex + 1} of 5`;

            document.getElementById('practiceWeakSentence').innerText = `"${activeKata.drills[drillIndex]}"`;

            document.getElementById('practiceInput').value = "";

            document.getElementById('inputArea').classList.remove('hidden');

            document.getElementById('practiceSuccessArea').classList.add('hidden');

            setTimeout(() => document.getElementById('practiceInput').focus(), 100);

        }


        function forge() {

            const val = document.getElementById('practiceInput').value.trim();

            if (!val) return;

            const slash = document.getElementById('slashEffect');

            slash.classList.add('animate-slash');

            setTimeout(() => {

                slash.classList.remove('animate-slash');

                document.getElementById('practiceWeakSentence').innerHTML = `<span class="forged-text">"${val}"</span>`;

                document.getElementById('inputArea').classList.add('hidden');

                document.getElementById('practiceSuccessArea').classList.remove('hidden');

                streak++;

                document.getElementById('streakCounter').innerText = streak;

            }, 400);

        }


        function nextPractice() {

            drillIndex++;

            if (drillIndex < 5) loadDrill();

            else showMaster();

        }


        function showMaster() {

            document.getElementById('practiceStage').classList.add('hidden');

            document.getElementById('masterStage').classList.remove('hidden');

            document.getElementById('paragraphContent').innerHTML = activeKata.paragraph;

            const cover = document.getElementById('ninjaCover');

            cover.classList.remove('hidden');

            setTimeout(() => {

                cover.classList.add('animate-slice-open');

                setTimeout(() => {

                    cover.classList.add('hidden');

                    document.getElementById('masterInputArea').classList.remove('hidden');

                    document.getElementById('masterInput').focus();

                }, 2500);

            }, 500);

        }


        function forgeMaster() {

            const val = document.getElementById('masterInput').value.trim();

            if (!val) return;

            const target = document.getElementById('p-target');

            

            // Apply high contrast emerald style

            target.innerHTML = val;

            target.className = "bg-emerald-500 text-slate-950 font-black italic px-2 py-1 rounded-lg shadow-[0_0_15px_rgba(16,185,129,0.4)] transition-all";

            

            document.getElementById('masterInputArea').classList.add('hidden');

            document.getElementById('completionArea').classList.remove('hidden');

        }


        window.onload = renderGrid;

    </script>
